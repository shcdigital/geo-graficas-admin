// ---------- Editor ----------
function slugFrom(str){
  return (str||"").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,"")
    .replace(/[^a-z0-9]+/g,"-").replace(/(^-|-$)/g,"");
}
function esc(v){
  return String(v==null?"":v)
    .replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;").replace(/'/g,"&#39;");
}
function jsStr(v){
  return String(v==null?"":v).replace(/\\/g,"\\\\").replace(/'/g,"\\'");
}
function quote(v){
  const s=String(v).trim();
  if(/[:#\[\]{},&*!|>'"]|\s/.test(s)) return `"${s.replace(/"/g,'\\"')}"`;
  return s;
}
function sQuote(v){
  return `"${String(v).replace(/"/g,'\\"')}"`;
}
function listYaml(arr){
  return arr.length ? " "+JSON.stringify(arr.map(s=>String(s))) : " []";
}
function currentSlug(){ return editandoSlug || slugFrom($val("title")||"cuadernillo"); }

}

function buildMarkdown(){
  const precio=$val("precio")||"Consultar";
  const lines=[
    "---",
    `title: ${sQuote($val("title"))}`,
    `description: ${sQuote($val("description"))}`,
    `pubDate: ${$val("pubDate")}`,
    `nivel: "${$val("nivel")}"`,
    `ciclo: ${sQuote($val("ciclo"))}`,
    `asignatura: ${sQuote($val("asignatura"))}`,
    `materia: ${sQuote($val("materia"))}`,
    `formato: "PDF"`,
    `paginas: ${Number($("paginas").value)}`,
    `unidades:${listYaml(lists.unidad)}`,
    `incluye:${listYaml(lists.incluye)}`,
    `precio: ${sQuote(precio)}`,
    `precioDestacado: ${$("precioDestacado").checked}`,
    `coverFrom: "${palFrom}"`,
    `coverTo: "${palTo}"`,
    `emoji: "${$val("emoji")||"📘"}"`,
    `destacado: ${$("destacado").checked}`,
    `draft: ${$("draft").checked}`,
    ...(imagenRel()?[`imagen: /${imagenRel()}`]:[]),
    "---",
    "",
    $val("body")||"Escribí acá la descripción larga en Markdown…",
    "",
  ];
  return lines.join("\n");
}

function validate(){
  let ok=true;
  ["title","ciclo","asignatura","materia","paginas"].forEach(id=>{
    const bad=!$val(id);
    $(id).classList.toggle("invalid",bad);
    if(bad)ok=false;
  });
  const desc=$("description");
  const descBad=desc.value.trim().length<20;
  desc.classList.toggle("invalid",descBad);
  $("desc-error").hidden=!descBad;
  if(descBad)ok=false;
  const pag=Number($("paginas").value);
  if(!$val("paginas")||!Number.isInteger(pag)||pag<=0){
    $("paginas").classList.add("invalid");ok=false;
  }
  return ok;
}

async function subir(){
  if(!validate()){toast("⚠ Revisá los campos marcados en rojo");return;}
  if(!editandoSlug)$("pubDate").value=new Date().toISOString().split("T")[0];
  const slug=currentSlug();
  $("btnSubir").disabled=true;
  $("btnSubir").textContent="Subiendo…";
  try{
    if(imagenData){
      const up=await api("/api/imagen",{method:"POST",body:JSON.stringify({slug,ext:imagenData.ext,base64:imagenData.base64})});
      if(!up.ok)throw new Error(up.message||"No se pudo subir la imagen");
      imagenUrl="/img/recursos/"+slug+"."+imagenData.ext;
      imagenData=null;
    }
    const md=buildMarkdown();
    const msg=editandoSlug?`Actualizar ${slug}.md`:`Crear ${slug}.md`;
    const d=await api("/api/recurso",{method:"POST",body:JSON.stringify({slug,content:md,message:msg})});
    if(d.ok||d.message){
      if(editandoSlug)okPopup("Cambios guardados","Se actualizó “"+$("title").value.trim()+"”. Se verá reflejado en el sitio en unos minutos.");

  }catch(e){
    toast(e.message||"No se pudo publicar");
    $("btnSubir").disabled=false;
    $("btnSubir").textContent="Subir publicación";
  }
}

function stripQuotes(v){
  if(v===null||v===undefined)return v;
  let s=String(v).trim();
  if(s.startsWith('"')&&s.endsWith('"'))s=s.slice(1,-1);
  if(s.startsWith("'")&&s.endsWith("'"))s=s.slice(1,-1);
  return s;
}

function editar(slug){
  editandoSlug=slug;
  mostrarEditor("Editar: "+slug);
  $("btnBorrar").classList.remove("hidden");
  api("/api/recurso?slug="+encodeURIComponent(slug)).then((rec)=>{
    const md=rec.content||"";
    let fm={},body="";
    const m=md.match(/^---\n([\s\S]*?)\n---\n?([\s\S]*)$/);
    if(m){
      body=(m[2]||"").trim();
      m[1].split("\n").forEach(line=>{
        const kv=line.match(/^([a-zA-Z0-9_]+):\s*(.*)$/);
        if(!kv)return;
        const key=kv[1],val=kv[2];
        if(key==="unidades"||key==="incluye"){
          const listKey=key==="unidades"?"unidad":"incluye";
          let items=[];
          const v=(val||"").trim();
          if(v.startsWith("[")){
            try{items=JSON.parse(v).map(String);}catch(e){items=[];}
          }else{
            const block=(md.match(new RegExp("^"+key+":(?:\\n  - .*)+", "m"))||[""])[0];
            items=block.split("\n").slice(1).map(l=>l.replace(/^\s*-\s*/,"").trim()).filter(Boolean);
          }
          lists[listKey]=items.map(stripQuotes);
          renderChips(listKey);
          return;
        }
        fm[key]=stripQuotes(val);
      });
    }
    if(fm.title!==undefined)$("title").value=fm.title;
    if(fm.description!==undefined)$("description").value=fm.description;
    if(fm.nivel)$("nivel").value=fm.nivel;
    if(fm.ciclo!==undefined)$("ciclo").value=fm.ciclo;
    if(fm.asignatura!==undefined)$("asignatura").value=fm.asignatura;
    if(fm.materia!==undefined)setMateria(fm.materia);
    if(fm.paginas!==undefined)$("paginas").value=fm.paginas;
    if(fm.precio!==undefined)setPrecioSelect(fm.precio==="Consultar"?"Consultar":fm.precio);
    if(fm.pubDate!==undefined)$("pubDate").value=fm.pubDate;
    if(fm.emoji)$("emoji").value=fm.emoji;
    if(fm.coverFrom&&fm.coverTo)selectPalette(fm.coverFrom,fm.coverTo);
    $("destacado").checked=fm.destacado==="true"||fm.destacado===true;
    $("draft").checked=fm.draft==="true"||fm.draft===true;
    $("precioDestacado").checked=fm.precioDestacado==="true"||fm.precioDestacado===true;
    $("body").value=body||"";
    $("slug").value=slug;
    imagenData=null;
    imagenUrl=fm.imagen||"";
    if(imagenUrl){
      mostrarPortada();
      updatePreview();
      toast("Cuadernillo cargado");
    }else{
      detectarPortada(slug).then(()=>{
        updatePreview();
        toast("Cuadernillo cargado");
      });
    }
  }).catch(e=>{toast(e.message||"No se pudo cargar");});
}

async function detectarPortada(slug){
  const exts=["png","jpg","jpeg","webp","gif"];
  for(const ext of exts){
    try{
      const probe=portadaUrl(slug,ext);
      const res=await fetch(probe,{method:"HEAD",cache:"no-store"});
      if(res.ok){
        imagenUrl="/img/recursos/"+slug+"."+ext;
        mostrarPortada();
        return;
      }
    }catch(e){}
  }
}

    img.src="data:image/"+imagenData.ext+";base64,"+imagenData.base64;

  }else if(imagenUrl){
    img.src=portadaSrc();
    row.hidden=false;
  }else{
    row.hidden=true;
    img.removeAttribute("src");
  }
  updatePreview();
}

  if(imagenUrl)return imagenUrl.replace(/^\/+/, "");

}
